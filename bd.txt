CREATE DATABASE IF NOT EXISTS AgronovaMVC;
USE AgronovaMVC;

-- Eliminar tablas en orden para evitar problemas de llaves foráneas
DROP TABLE IF EXISTS compras;
DROP TABLE IF EXISTS pedidos;
DROP TABLE IF EXISTS productos;
DROP TABLE IF EXISTS clientes;
DROP TABLE IF EXISTS usuarios;

-- ========================================================
-- CREACIÓN DE TABLAS
-- ========================================================
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre_completo VARCHAR(60) NOT NULL,
    username VARCHAR(20) NOT NULL UNIQUE,
    password VARCHAR(20) NOT NULL,
    rol VARCHAR(15) NOT NULL
);

CREATE TABLE clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    nombre_completo VARCHAR(60) NOT NULL,
    ci_nit VARCHAR(15) NOT NULL,
    telefono VARCHAR(12),
    direccion VARCHAR(100) NOT NULL,
    zona VARCHAR(20) NOT NULL
);

CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre_insumo VARCHAR(50) NOT NULL,
    categoria VARCHAR(20) NOT NULL,
    precio_compra DECIMAL(8,2) NOT NULL,      -- Control del docente
    precio_venta DECIMAL(8,2) NOT NULL,       -- Diferenciado de precio_compra
    stock_disponible INT NOT NULL,
    fecha_vencimiento VARCHAR(15) NOT NULL
);

CREATE TABLE pedidos (
    id_pedido INT AUTO_INCREMENT PRIMARY KEY,
    id_cliente INT NOT NULL,
    id_producto INT NOT NULL,
    cantidad INT NOT NULL,
    total_pagar DECIMAL(8,2) NOT NULL,
    estado_entrega VARCHAR(15) NOT NULL,
    fecha_registro VARCHAR(20) NOT NULL,
    FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente),
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto)
);

CREATE TABLE compras (
    id_compra INT AUTO_INCREMENT PRIMARY KEY,
    id_producto INT NOT NULL,
    cantidad INT NOT NULL,
    total_costo DECIMAL(8,2) NOT NULL,
    fecha_compra VARCHAR(20) NOT NULL,
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto)
);
